    <script>
        // Load YouTube IFrame API asynchronously
        var tag = document.createElement('script');
        tag.src = "https://www.youtube.com/iframe_api";
        var firstScriptTag = document.getElementsByTagName('script')[0];
        firstScriptTag.parentNode.insertBefore(tag, firstScriptTag);

        let player;
        let isPlaying = false;

        function onYouTubeIframeAPIReady() {
            player = new YT.Player('yt-player', {
                height: '0',
                width: '0',
                videoId: 'gIjZ8zcg-m0', // Hussein Al Deek - Ya Aaybo
                playerVars: {
                    'playsinline': 1,
                    'controls': 0,
                    'disablekb': 1,
                    'autoplay': 0
                },
                events: {
                    'onStateChange': onPlayerStateChange
                }
            });
        }

        function onPlayerStateChange(event) {
            const playIcon = document.getElementById('play-icon');
            const playingOverlay = document.getElementById('playing-overlay');

            if (event.data === YT.PlayerState.PLAYING) {
                isPlaying = true;
                playIcon.className = 'fas fa-pause text-lg';
                playingOverlay.classList.remove('hidden');
            } else {
                isPlaying = false;
                playIcon.className = 'fas fa-play ml-0.5 text-lg';
                playingOverlay.classList.add('hidden');
            }
        }

        function toggleMusic() {
            if (!player || typeof player.playVideo !== 'function') {
                showToast("Loading YouTube audio...");
                return;
            }

            if (isPlaying) {
                player.pauseVideo();
            } else {
                player.playVideo();
                showToast("Playing 'Ya Aaybo' by Hussein Al Deek 🎵");
            }
        }

        // --- Press & Hold Heart Animation logic ---
        let heartCount = 0;
        let holdInterval;
        const heartBtn = document.getElementById('heart-btn');
        const counterText = document.getElementById('heart-counter-text');

        function startHold(e) {
            if (e.cancelable) e.preventDefault();
            
            // Initial pop
            createFloatingHeart();
            heartCount++;
            updateCounter();

            holdInterval = setInterval(() => {
                heartCount++;
                updateCounter();
                createFloatingHeart();
            }, 120);
        }

        function stopHold() {
            clearInterval(holdInterval);
        }

        function updateCounter() {
            counterText.innerHTML = `<span class="text-accent-pink font-semibold">${heartCount}</span> hearts sent on their way ❤️`;
        }

        function createFloatingHeart(emoji = '❤️') {
            const rect = heartBtn.getBoundingClientRect();
            const heart = document.createElement('div');
            heart.className = 'floating-heart';
            heart.innerHTML = emoji;
            
            // Random horizontal drift and rotation
            const randomX = (Math.random() - 0.5) * 120;
            const randomR = (Math.random() - 0.5) * 40;
            
            heart.style.setProperty('--tw-random-x', `${randomX}px`);
            heart.style.setProperty('--tw-random-r', `${randomR}deg`);
            
            // Position at center of button
            heart.style.left = `${rect.left + rect.width / 2}px`;
            heart.style.top = `${rect.top + rect.height / 2}px`;
            heart.style.fontSize = `${20 + Math.random() * 12}px`;

            document.body.appendChild(heart);

            setTimeout(() => {
                heart.remove();
            }, 1200);
        }

        function triggerEmojiReaction(emoji) {
            for (let i = 0; i < 6; i++) {
                setTimeout(() => {
                    createFloatingHeart(emoji);
                }, i * 80);
            }
            showToast(`Sent reaction: ${emoji}`);
        }

        // Attach touch and mouse listeners
        heartBtn.addEventListener('mousedown', startHold);
        window.addEventListener('mouseup', stopHold);

        heartBtn.addEventListener('touchstart', startHold, { passive: false });
        window.addEventListener('touchend', stopHold);

        // Toast Message Notification Helper
        function showToast(msg) {
            const toast = document.getElementById('toast-message');
            const toastText = document.getElementById('toast-text');
            toastText.innerText = msg;
            
            toast.classList.remove('opacity-0', 'translate-y-4', 'pointer-events-none');
            toast.classList.add('opacity-100', 'translate-y-0');

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-4', 'pointer-events-none');
                toast.classList.remove('opacity-100', 'translate-y-0');
            }, 2500);
        }

        function showMessage(msg) {
            showToast(msg);
        }

        function copyReplyLink() {
            const dummy = document.createElement('input');
            document.body.appendChild(dummy);
            dummy.value = window.location.href;
            dummy.select();
            document.execCommand('copy');
            document.body.removeChild(dummy);
            showToast("Copied card link! Send it back to Jake.");
        }

        function copyCardLink() {
            copyReplyLink();
        }
